"use client";
import apiaddress from "@/apirequests/apiaddress";
import Breadcrumb from "@/components/Breadcrumbs/Breadcrumb";
import { ToastContainer } from "react-toastify";
import "react-toastify/dist/ReactToastify.css";
import { toastParser } from "@/apirequests/getcustomersbyshopid";
import { useState } from "react";
import { useGlobalState } from "@/js/globaluser";

const Page = () => {
  const {user} = useGlobalState()
  const [file, setFile] = useState(null);
  const token = localStorage.getItem("token")
  const [message, setMessage] = useState("");
  const [loading, setLoading] = useState(false); // State for loader

  const handleFileChange = (e) => {
    setFile(e.target.files[0]);
  };

  const handleSubmit = async (e) => {
    e.preventDefault();

    if (!file) {
      setMessage("Please select a file.");
      return;
    }

    const formData = new FormData();
    formData.append("backupFile", file);

    setLoading(true);
    try {
      const res = await fetch(apiaddress + "/email/restoredata", {
        method: "POST",
        headers:{
          token
        },
        body: formData,
      });

      const data = await res.json();
      if (data.success) {
        setMessage("Database restored successfully.");
      } else {
        setMessage("Failed to restore database. " + data.message);
      }
    } catch (error) {
      console.error("Error uploading file:", error);
      setMessage("An error occurred while restoring the database.");
    } finally {
      setLoading(false);
    }
  };

  const sendBackup = async () => {
    setLoading(true);
    try {
      let data = await fetch(apiaddress + "/email/sendbackupemail",{
        method:"GET",
        headers:{
          token
        }
      });
      let parsed = await data.json();
      const next = () => {
        console.log("hello");
      };
      await toastParser(parsed, next);
    } catch (error) {
      console.error("Error sending backup:", error);
      setMessage("An error occurred while sending the backup.");
    } finally {
      setLoading(false);
    }
  };

  return (
    <>

        <ToastContainer />
        <div className="mx-auto max-w-270">
          <Breadcrumb pageName="Database Backup" />
        </div>
        <div className="flex items-center justify-center py-30 bg-boxdark">
          <div className="bg-black p-8 rounded-xl shadow-2xl transform hover:scale-105 transition-transform duration-300">
            {loading ? (
              <div>
                <div className="flex justify-center items-center mb-8">
                  <div className="loader border-t-4 border-b-4 border-blue-500 rounded-full w-12 h-12 animate-spin"></div>
                </div>
              </div>
            ) : (
              <div className="mb-8">
                <div className="flex justify-between space-x-4">
                  <button
                    className="border-green-500 border-2 w-full rounded-md shadow-lg p-5 text-white font-bold text-xl hover:scale-110 transform transition-transform duration-200 bg-gradient-to-r from-green-500 to-green-700"
                    onClick={sendBackup}
                  >
                    Send Backup Email
                  </button>
                </div>

                <div className="mt-8">
                  <form onSubmit={handleSubmit} className="flex flex-col items-center">
                    <input
                      type="file"
                      className="border-red-500 border-2 rounded-md shadow-lg p-3 bg-gradient-to-r from-red-500 to-red-700 text-white font-bold text-lg cursor-pointer hover:scale-105 transition-transform duration-200"
                      onChange={handleFileChange}
                      accept=".txt"
                      required
                    />
                    <button
                      type="submit"
                      className="border-rose-500 w-full border-2 rounded-md shadow-lg p-3 mt-6 bg-gradient-to-r from-rose-500 to-rose-700 text-white font-bold text-lg hover:scale-110 transform transition-transform duration-200"
                    >
                      Restore
                    </button>
                  </form>
                </div>
                
              </div>
            )}

            <div className="mt-8">
              {message && (
                <p
                  className={`mt-4 text-center ${
                    message.includes("successfully") ? "text-blue-400" : "text-red-400"
                  } font-semibold animate__animated animate__fadeIn`}
                >
                  {message}
                </p>
              )}
            </div>
          </div>
        </div>
    </>
  );

};

export default Page;
